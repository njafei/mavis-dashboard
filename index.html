// ── Countdown helpers ── function pad(n) { return String(n).padStart(2,'0'); } function getNextTime(h, m) { const now = new Date(); const next = new Date(now); next.setHours(h, m, 0, 0); if (next <= now) next.setDate(next.getDate() + 1); const diff = Math.round((next - now) / 1000); const h2 = Math.floor(diff / 3600); const m2 = Math.floor((diff % 3600) / 60); return pad(h2) + 'h' + pad(m2) + 'm'; } function getNextMusic() { const now = new Date(); const mins = now.getMinutes(); const secs = now.getSeconds(); const nextSlot = mins < 30 ? 30 : 60; const diff = ((nextSlot - mins) * 60 - secs) % (60 * 60); const h2 = Math.floor(diff / 3600); const m2 = Math.floor((diff % 3600) / 60); return pad(h2) + 'h' + pad(m2) + 'm'; } function getNextHourly() { const now = new Date(); const secs = now.getSeconds(); const diff = 3600 - (now.getMinutes() * 60 + secs) % 3600; const h2 = Math.floor(diff / 3600); const m2 = Math.floor((diff % 3600) / 60); return pad(h2) + 'h' + pad(m2) + 'm'; } function updateTimers() { const el = document.getElementById('next-music'); if (el) el.textContent = getNextMusic() + ' 后'; const el2 = document.getElementById('next-stock'); if (el2) el2.textContent = getNextHourly() + ' 后'; const el3 = document.getElementById('next-watchdog'); if (el3) el3.textContent = getNextHourly() + ' 后'; } window.addEventListener('DOMContentLoaded', updateTimers);
